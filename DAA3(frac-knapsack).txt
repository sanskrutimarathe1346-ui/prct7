**Write a program to solve a fractional Knapsack
problem using a greedy method.**

public class FractionalKnapsack {
    public static void main(String[] args) {
        int n = 5;
        double capacity = 60;
        int[] weight = {5, 10, 15, 22, 25};
        int[] value = {30, 40, 45, 77, 90};
        int[] item = {1, 2, 3, 4, 5};
        double[] ratio = new double[n];
        
        for (int i = 0; i < n; i++) 
        {
            ratio[i] = (double) value[i] / weight[i];
        }
        
        for (int i = 0; i < n - 1; i++) 
        {
            for (int j = i + 1; j < n; j++) 
            {
                if (ratio[i] < ratio[j])
                 {
                    double temp = ratio[i];
                    ratio[i] = ratio[j];
                    ratio[j] = temp;
                    int t = weight[i];
                    weight[i] = weight[j];
                    weight[j] = t;
                    t = value[i];
                    value[i] = value[j];
                    value[j] = t;
                    t = item[i];
                    item[i] = item[j];
                    item[j] = t;
                }
            }
}
        double totalValue = 0;
        System.out.println("Selected Items:");
        for (int i = 0; i < n; i++)
         {
            if (capacity == 0)
                break;
            if (weight[i] <= capacity)
             {
                capacity -= weight[i];
                totalValue += value[i];
                System.out.println("Item " + item[i]
                        + " - Fully selected");
            } 
            else
             {
                double fraction = capacity / weight[i];
                totalValue += value[i] * fraction;
                System.out.println("Item " + item[i]
                        + " - Fraction selected: "
                        + fraction);
                capacity = 0;
            }
        }
        System.out.println("Maximum Value: " + totalValue);
    }
}
