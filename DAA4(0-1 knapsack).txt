**Write a program to solve a 0-1 Knapsack problem
using dynamic
programming or branch and bound strategy.**

 public class Knapsack 
    {
    public static void main(String[] args) 
    {
    int[] w = {2, 3, 4, 5};
    int[] v = {3, 4, 5, 6};
    int capacity = 5;
    int n = w.length;
    int[][] dp = new int[n + 1][capacity + 1];
    
    for (int i = 1; i <= n; i++) 
    {
        for (int j = 1; j <= capacity; j++) 
        {
            if (w[i - 1] <= j)
             {
                dp[i][j] = Math.max(
                    v[i - 1] + dp[i - 1][j - w[i - 1]],
                    dp[i - 1][j]
                );
            } 
            else
             {
                dp[i][j] = dp[i - 1][j];
            }
        }
    }
    int j = capacity;
    System.out.print("Selected Items: ");
    for (int i = n; i > 0; i--)
     {
        if (dp[i][j] != dp[i - 1][j]) 
        {
            System.out.print(i + " ");
            j -= w[i - 1];
        }
    }
System.out.println();
System.out.println("Maximum Value = " + dp[n][capacity]);
}
}
